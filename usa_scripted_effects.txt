#spread corruption through trade
on_monthly_pulse = {
	if = {
		limit = {
			all_country = {
				any_active_trade_node = {
					any_trade_node_member_country = {
						has_government_mechanic = corruption_scaling
					}
				}
			}
		}
		add_corruption = 0.03
	}
}



#corrupted center of trade victim activate
on_monthly_pulse = {
	if = {
		limit = {
			any_active_trade_node = {
	    		all_trade_node_member_province = {
					and = {
						or = {
							province_has_center_of_trade_of_level = 1
					    	is_trade_node_province = yes
						}
						not = { has_province_modifier = detrimental_corruption }

		    			owner = {
			    			and = {
				    			not = { has_government_mechanic = corruption_scaling }
					    		corruption = 3
						    }
		    			}
					}
	    		}
    		}
		}
		remove_province_modifier = beneficial_corruption
		add_province_modifier = {
			name = detrimental_corruption
			duration = -1
		}
	}
}

#corrupted center of trade switch off
on_monthly_pulse = {
	if = {
		limit = {
			any_active_trade_node = {
	    		all_trade_node_member_province = {
					and = {
						or = {
							province_has_center_of_trade_of_level = 1
					    	is_trade_node_province = yes
						}
						or = {
							has_province_modifier = beneficial_corruption
							has_province_modifier = detrimental_corruption
						}

		    			owner = {
			    			and = {
				    			not = { has_government_mechanic = corruption_scaling }
					    		not = { corruption = 3 }
						    }
		    			}
					}
	    		}
    		}
		}
		remove_province_modifier = detrimental_corruption
		remove_province_modifier = beneficial_corruption
	}
