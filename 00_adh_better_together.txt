#east stl breakaway
country_event = {
    id = better_together.4
    title = better_together.4.t
    desc = better_together.4.d
    fire_only_once = yes

    is_triggered_only = yes

    immediate = {
        hidden_effect = {
            release = ESL
            ESL = {
                change_government = republic
                add_government_reform = midwestern_trade_hub
                adopt_reform_progress = root
                capital_scope = {
                    build_to_forcelimit = {
                        infantry = 0.6
                        cavalry = 0.1
                        artillery = 0.3
                    }
                }
            }
        }
        
    }

    option = {
        name = better_together.4.a
        add_prestige = -25
        every_province = {
            limit = {
                owned_by = STL
                region = region_mo
            }
            add_unrest = 2
        }
        add_power_projection = {
            type = humiliated_by_rival
            amount = -30
        }
    }

    option = {
        name = better_together.4.b
        switch_tag = ESL
        STL = {
            declare_war_with_cb = {
                casus_belli = cb_annex
                who = ESL
            }
        }
    }    

    option = {
        name = better_together.4.c
        add_mil_power = -75
        every_province = {
            limit = {
                owned_by = STL
                region = region_mo
            }
            add_unrest = 1.5
        }
        declare_war_with_cb = {
			casus_belli = cb_annex
			who = ESL
		}
    }
}
