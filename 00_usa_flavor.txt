    	        	}
        		}
			}
			add_country_modifier = {
				name = core_bloc_member
				duration = -1
			}
		}
		#demote isolate bloc cores
		if = {
		    limit = {
        		has_country_modifier = core_bloc_member
		        NOT = {
        		    any_neighbor_country = {
                		OR = {
	    	                tag = overlord
							and = {
								has_country_modifier = core_bloc_member
								overlord = {
            				        any_subject_country = {
			                	        tag = ROOT
                    				}
            				    }
							}
						}
				    }
      			}
	    	}
    		remove_country_modifier = core_bloc_member
		}
		if = { #remove noncontiguous bloc members
			limit = {
				is_subject_of_type = midwest_bloc_subordinate
				not = {
					or = {
						AND = {
				            any_neighbor_country = {
								overlord = {
            				        any_subject_country = {
			                	        tag = ROOT
                    				}
            				    }
            				}
    			    	}
						any_neighbor_country = {
							overlord_of = root
						}
					}
					
				}
				
			}
			grant_independence = yes
		}
	}

country_event = {
	id = flavor_usa.1905
	title = flavor_usa.1905.t
	desc = flavor_usa.1905.desc
	mean_time_to_happen = {	months = 50 }

	trigger = {
		any_trade_node = {
			any_province = {
				has_colonist = yes
			}
			any_province = {
				not = { has_colonist = yes }
				is_city = no
				any_neighbor_province = {
					culture = prev
					owner = {
						or = {
							has_reform = steppe_horde
							has_reform = great_mongol_state_reform
							has_reform = bismarck_steppe_horde
						}
					}
				}
			}
		}
	}

	immediate = { #match to trigger to prevent blank options
		random_province = {
			limit = {
				not = { has_colonist = yes }
				is_city = no
				any_neighbor_province = {
					culture = prev
					owner = {
						and = {
							tag = ROOT
							or = {
								has_reform = steppe_horde
								has_reform = great_mongol_state_reform
								has_reform = bismarck_steppe_horde
							}
						}
						
					}
				}
			}
			save_event_target_as = threatened_province
		}
	}


	option = { #protect the frontier
		name = flavor_usa.1905.a
		event_target:threatened_province = {
			create_colony = 1000
			add_territorial_core = ROOT
		}	
		add_manpower = -20
		change_government_reform_progress = -15
	}

	option = { #we lack the resources
		name = flavor_usa.1905.b
		every_country = {
			limit = {
				has_reform = steppe_horde
				has_reform = bismarck_steppe_horde
			}
			reverse_add_opinion = {
				who = root
				modifier = horde_refuse_aid
			}
		}
	}
