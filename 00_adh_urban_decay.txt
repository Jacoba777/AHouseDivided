midwest_urban_decay = {

    potential = {
        any_owned_province = {
            development = 30
        }
        NOT = { has_country_modifier = urban_acclimatization }
		not = { has_government_mechanic = perfectionism_mechanic }
		not = {
			and = {
				is_part_of_hre = yes
				emperor = {
					has_country_flag = trifecta_emperor
				}
			}
		}
        num_of_owned_provinces_with = {
            value = 3
            has_province_modifier = merchant_flight
        }
    }

    can_start = {
        not = { stability = 0 }
        OR = {
            not = { treasury = 0 }
            num_of_loans = 5
        }
        any_owned_province = {
            development = 20
            unrest = 2
        }
    }

    can_stop = {
        stability = 2
        any_owned_province = {
            development = 20
            NOT = { unrest = 1 }
        }
    }

    progress = {
        modifier = {
            factor = 1.5
            not = { stability = 0 }
        }
        modifier = {
            factor = 1.25
            num_of_loans = 10
        }
        modifier = {
            factor = -0.75
            years_of_income = 3
        }
        modifier = {
            factor = 1.5
            any_owned_province = {
                development = 20
                unrest = 3
            }
        }
    }

    can_end = {
        and = {
			stability = 2
	        years_of_income = 2
			not = { num_of_loans = 3}
		
			or = {
    	    	any_owned_province = {
	    	        development = 20
					or = {
						NOT = { unrest = 1 }
						not = { has_province_modifier = urban_blight }
					}
	        	}
				not = {
					any_owned_province = {
						development = 17
					}
				}
			}
		}
    }

    modifier = {
        global_tax_modifier = -0.2
        production_efficiency = -0.15
        trade_efficiency = -0.15
        global_unrest = 2
        development_cost = 0.1
    }

    on_start = urban_decay.1
    on_end = urban_decay.6

    on_monthly = {
        random_events = {
            500 = 0
            100 = urban_decay.2   # dev drain  = {}
            10 = urban_decay.3   # infrastructure  = {}
            35 = urban_decay.4   # capital flight = {}
			40 = urban_decay.5   # social unrest = {}
        }
    }

    ended_by_country_breaking_to_rebels = no
}
