		}
		provinces_to_highlight = {
			OR = {
				area = area_mn_faribault
				area = area_mn_south_minneapolis
			}
			OR = {
				NOT { country_or_non_sovereign_subject_holds = ROOT }
				devastation = 1
				unrest = 10
				local_autonomy = 50
			}
		}
	}
