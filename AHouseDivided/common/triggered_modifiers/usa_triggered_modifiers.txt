# In my testing, the AI only seems to colonize if they have an income of at least 10 ducats.
# This modifier intends to trick the AI into thinking its income is higher.
triggered_modifier_ai_colonization_incentive = {
    potential = {
        ai = yes
    }
    
	# Grant the bonus if the country is AI, has a colonist but no active colony, and has no neighbors
    trigger = {
        ai = yes
		NOT = { any_owned_province = { is_colony = yes } }
		NOT = { any_neighbor_country = { exists = yes } }
		num_of_colonists = 1
    }
    
    global_tax_income = 120
}

#ADH modifiers

ak_colony_debuff = {
	potential = {
			always = yes
	}

	trigger = {
		any_owned_province = {
			and = {
				is_colony = yes
				region = region_ak
				owner = {
					OR = {
						NOT = { primary_culture = culture_alaskan }
						NOT = { primary_culture = culture_panhandle_alaskan }			
					}
				}				
			}
		}
	}
	ai_weight = {
		factor = -1000 # Do what you can to avoid this modifier
	}

	colony_cost_modifier = 6
	native_uprising_chance = 1.5
	colonist_placement_chance = -0.95
	global_colonial_growth = -150
	native_assimilation = -0.95
}
