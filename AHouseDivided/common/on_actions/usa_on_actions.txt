on_monthly_pulse = {
    if = {
        limit = {
            has_government_mechanic = midwest_mercantilism
        }
        country_event = { #midwest tier assignments updater
            id = flavor_usa.1401
            days = -1
        }
        country_event = { #midwest core member updater
            id = flavor_usa.1402
            days = -1
        }
    }
    if = {#chicago outfit assassination event
        limit = {
            has_reform = chicago_outfit
        }
        country_event = { 
            id = flavor_usa.1451
            days = -1
        }
    }
    if = {
        limit = {
            has_country_modifier = anti_corp_stance
        }
        remove_country_modifier = vancouver_corporation_investing
    }
    
}

#corporatocracy founding (and midwest flag setting)
on_startup = {
    every_country = {
        limit = {
            not = { has_reform = united_cantons_reform }
            not = { has_reform = reformer_state_reform }
            not = { num_of_cities = 2 }
            not = { has_country_flag = corpo_freedom }
            has_reform = steppe_horde
            any_core_province = {
                or = {
                    trade_goods = coal
                    trade_goods = copper
                    trade_goods = gold
                    trade_goods = gems
                }
            }
            
        }
        change_government = republic
        add_government_reform = corporatocracy
    }
    country_event = {
        id = flavor_usa.1401
        days = -1
    }
    FBT = {
        country_event = {
            id = flavor_usa.1405
            days = 20
        }
    }
    
}

#change clients to corporatocracies
on_create_client_state = {
    if = {
        limit = {
            overlord = {
                has_reform = corporatocracy
            }
        }
        change_government = republic
        add_government_reform = corporatocracy
    }
}

#vancorp anti corpo stance flagging
on_war_won = {
    if = {
        limit = {
            from = {
                has_reform = stadthalter_monarchy_reform
            }
        }
        root = {
            add_country_modifier = {
                name = anti_corp_stance
                duration = 7300
            }
            remove_country_modifier = vancouver_corporation_investing
            remove_country_modifier = vancouver_begrudging_dependency
            remove_country_modifier = vancouver_corporate_dependency
            remove_country_modifier = vancouver_corporate_fear
            remove_country_modifier = vancouver_integrated_market
            remove_country_modifier = vancouver_recent_investment
            remove_country_modifier = vancouver_recent_corporate_dependency
            remove_country_modifier = vancouver_recent_integrated_market
        }
    }
}

#natural anti-corporate stance flagging, for flavor and to prevent abuse of vancorp
on_yearly_pulse = {
    if = {
        limit = {
            NOT = { has_country_modifier = anti_corp_stance }
            any_neighbor_country = {
                or = {
                    has_country_flag = vancouver_corp_flag
                    overlord = {
                        has_country_flag = vancouver_corp_flag
                    }
                }
                root = {
                    not = {
                        has_opinion = {
                            who = prev
                            value = -150
                        }
                    }
                }
            }
        }
        add_country_modifier = {
            name = anti_corp_stance
            duration = 7300
        }
    }
    
}
