country_event = {
    id = flavor_usa.301
    title = flavor_usa.301.t
    desc = flavor_usa.301.desc
    picture = GFX_event_corruption

    is_triggered_only = yes
    
    trigger = {
        has_country_flag = newsouth
    }

    immediate = {
        root = {
            add_country_modifier = {
                name = dixie_corruption
                duration = 600
            }
        }

        # Find up to 4 qualifying Old South targets
        random_neighbor_country = {
            limit = {
                has_country_flag = oldsouth
                OR = {
                    not = {
                        has_opinion = {
                            who = ROOT

                    is_rival = ROOT
                    ROOT = { is_rival = PREV }
                }
            }
            save_event_target_as = flavor_usa_target_1_1
        }

                has_country_flag = oldsouth

                        has_opinion = {

                country_event = {
                    id = flavor_usa.45
                    days = -1
                }
            }
            ROOT = { add_treasury = -500 }
        }
    }
    option = {
        name = "Fund corruption in [flavor_usa_target_2_1.GetName]"
