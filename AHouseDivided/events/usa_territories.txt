# Department of Alaska
country_event = {
	id = flavor_xca.100
	title = flavor_xca.100.t
	desc = flavor_xca.100.desc
	picture = EYALET_eventPicture
	fire_only_once = yes
	trigger = {
		has_dlc = "Domination"
		or = {
			tag = XUS
			tag = XCA
		}
		not = {
			exists = XAK
		}
		capital_scope = {
			NOT = {
				region = region_ak
			}			#Why Eyalets when you can practically rule the stuff yourself?
		}
		num_of_owned_provinces_with = {
			region = region_ak
			value = 7
		}
	}
	mean_time_to_happen = {
		months = 12
	}
	immediate = {
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						exists = XAK
					}
				}
				XAK = {
					save_event_target_as = first_potential_eyalet
				}
			}
		}
	}
	option = {
		# Create the ak territory
		name = flavor_xca.100.a
		trigger = {
			has_saved_event_target = first_potential_eyalet
		}
		hidden_effect = {
			every_owned_province = {
				limit = {
					is_core = event_target:first_potential_eyalet
				}
				remove_core = event_target:first_potential_eyalet
			}
		}
		region_ak = {
			add_core = event_target:first_potential_eyalet
		}
		custom_tooltip = territory_claims_tt
		release = event_target:first_potential_eyalet
		create_subject = {
			subject_type = eyalet
			who = event_target:first_potential_eyalet
		}
		event_target:first_potential_eyalet = {
			if = {
				limit = {
					NOT = {
						government = republic
					}
				}
				change_government = republic
			}
			add_government_reform = eyalet_government
			add_liberty_desire = -25
			eyalet_effect = {
				tag = EY1
			}
		}
		add_dip_power = 75
		ai_chance = {
			factor = 1
		}
	}
	option = {
		# Fallback
		name = flavor_xca.210.d
		add_adm_power = 100
		ai_chance = {
			factor = 0
		}
	}
}
