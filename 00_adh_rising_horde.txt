#merchantile class demands more protection, but the government is too weak to provide it
country_event = {
    id = rising_horde.5
    title = rising_horde.5.t
    desc = rising_horde.5.d
	is_triggered_only = yes



    option = {
        name = rising_horde.5.a

        random_owned_province = {
            limit = {
                development = 20
            }
            add_province_modifier = {
                name = merchant_flight
                duration = 720
            }
        }

        add_prestige = -5
    }

    option = {
        name = rising_horde.5.b

        random_owned_province = {
            limit = {
                OR = {
                    any_neighbor_province = {
						owner = {
							OR = {
								has_reform = steppe_horde
								has_reform = great_mongol_state_reform
							}
						}
					}
                    province_has_center_of_trade_of_level = 1
                }
            }
            add_province_modifier = {
                name = militarized_trade_corridor
                duration = 720
            }
        }

        add_mil_power = 25
    }

    option = {
        name = rising_horde.5.c

        random_owned_province = {
            limit = {
                province_has_center_of_trade_of_level = 1
            }
            add_province_modifier = {
                name = merchant_protection_contracts
                duration = 720
            }
        }

        add_dip_power = -50
        add_prestige = 5
    }
}


#full wartime conversion to aggressive version of reform, only for those bordering the horde
country_event = {
    id = rising_horde.6
    title = rising_horde.6.t
    desc = rising_horde.6.d
    trigger = {
        has_country_flag = risinghordefour
		not = { has_country_modifier = emergency_war_funding }
		not = { has_country_modifier = military_auditing }

        any_neighbor_country = {
            OR = {
				has_reform = steppe_horde
				has_reform = great_mongol_state_reform
			}
        }
    }

    mean_time_to_happen = {
        months = 60

        modifier = {
            factor = 0.5
            any_neighbor_country = {
                OR = {
					has_reform = steppe_horde
					has_reform = great_mongol_state_reform
				}
                army_size = 20000
            }
        }

        modifier = {
            factor = 0.75
            stability = 0
        }
    }

    option = {
        name = rising_horde.6.a

        # Swap to aggressive reform
		if = {
			limit = {
				has_reform = midwestern_trade_hub
			}
			add_government_reform = midwestern_trade_hub_agressive
		}
		if = {
			limit = {
				has_reform = proscriptive_trade_hub
			}
			add_government_reform = proscriptive_trade_hub_aggressive
		}
		if = {
			limit = {
				has_reform = chicago
			}
			add_government_reform = chicago_aggressive
		}
		if = {
			limit = {
				has_reform = midwest_chartered_company_government
			}
			add_government_reform = midwest_chartered_company_government_aggressive
		}
    }

    option = {
        name = rising_horde.6.b

        add_stability = -1
        add_prestige = -5

        add_country_modifier = {
            name = stubborn_commercialism
            duration = 1825
        }
    }
}


#reversion of reform after the horde is defeated
country_event = {
    id = rising_horde.7
    title = rising_horde.7.t
    desc = rising_horde.7.d

    is_triggered_only = yes

    option = {
        name = rising_horde.7.a

        # Government normalization
        if = {
			limit = {
				has_reform = midwestern_trade_hub_agressive
			}
			add_government_reform = midwestern_trade_hub
		}
		if = {
			limit = {
				has_reform = proscriptive_trade_hub_aggressive
			}
			add_government_reform = proscriptive_trade_hub
		}
		if = {
			limit = {
				has_reform = chicago_aggressive
			}
			add_government_reform = chicago
		}
		if = {
			limit = {
				has_reform = midwest_chartered_company_government_aggressive
			}
			add_government_reform = midwest_chartered_company_government
		}

        # Economic rebound
        add_country_modifier = {
            name = postwar_boom
            duration = 3650
        }

		add_country_modifier = {
			name = postwar_structures
			duration = 7300
		}

        # Stability + prestige recovery
        add_stability = 1
        add_prestige = 15
    }
}
